// One step of a 1-D Kalman filter per fix: predict, then update. Change R or Q and run again.
let x = FIXES[0];        // state estimate: latitude [deg]
let P = 1e-4;            // estimate variance [deg^2]
const Q = 1e-8;          // process noise: drift of the true position per step [deg^2]
const R = 2.5e-6;        // measurement noise: GPS sigma about 1.6e-3 deg (175 m)

for (const z of FIXES.slice(1)) {
  P = P + Q;                     // predict: the state stays, the uncertainty grows
  const K = P / (P + R);         // gain: how far to trust the new fix
  x = x + K * (z - x);           // update with the innovation z - x
  P = (1 - K) * P;
  console.log(row(z, K, x, P));
}
// Web-map tiles: which tile holds a point, and its quadkey (x and y bits interleaved, Bing Maps scheme).
function tile(lat, lon, zoom) {
  const n = 1 << zoom, phi = lat * Math.PI / 180;
  return [Math.floor((lon + 180) / 360 * n), Math.floor((1 - Math.asinh(Math.tan(phi)) / Math.PI) / 2 * n)];
}
function quadkey(x, y, zoom) {
  let key = '';
  for (let i = zoom; i > 0; i--) {
    const mask = 1 << (i - 1);
    key += ((x & mask) ? 1 : 0) + ((y & mask) ? 2 : 0);   // a bit of x adds 1, a bit of y adds 2
  }
  return key;
}
const [x, y] = tile(21.03, 105.85, 12);                  // Hanoi at zoom 12
console.log('tile', x, y, 'quadkey', quadkey(x, y, 12));
console.log('x bits', x.toString(2).padStart(12, '0'), 'y bits', y.toString(2).padStart(12, '0'));
// check: the parent tile (x >> 1, y >> 1) has the same key without its last digit
console.log('parent ok:', quadkey(x >> 1, y >> 1, 11) === quadkey(x, y, 12).slice(0, -1));
