# Great circle against rhumb line (the straight line on a Mercator map), no numpy needed.
def great_circle(a, b):
    (p1, l1), (p2, l2) = [(math.radians(p), math.radians(l)) for p, l in (a, b)]
    h = math.sin((p2 - p1) / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin((l2 - l1) / 2) ** 2
    return 2 * R * math.asin(math.sqrt(h))       # haversine formula

def rhumb(a, b):
    (p1, l1), (p2, l2) = [(math.radians(p), math.radians(l)) for p, l in (a, b)]
    dl = (l2 - l1 + math.pi) % (2 * math.pi) - math.pi        # shorter way round
    dpsi = math.log(math.tan(math.pi / 4 + p2 / 2) / math.tan(math.pi / 4 + p1 / 2))
    q = (p2 - p1) / dpsi if abs(dpsi) > 1e-12 else math.cos(p1)
    return R * math.hypot(p2 - p1, q * dl)
