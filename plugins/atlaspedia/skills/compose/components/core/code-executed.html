import math

R = 6371.0088                                   # mean Earth radius in km (the sphere every module assumes)
CITIES = {"Hanoi": (21.03, 105.85), "Nuuk": (64.18, -51.72)}

def mercator(lat_deg, lon_deg, lon0_deg=0.0):
    """Spherical Mercator forward equations: x = R (lambda - lambda0), y = R ln tan(pi/4 + phi/2)."""
    lat, dlon = math.radians(lat_deg), math.radians(lon_deg - lon0_deg)
    return R * dlon, R * math.log(math.tan(math.pi / 4 + lat / 2))

for name, (lat, lon) in CITIES.items():
    x, y = mercator(lat, lon)
    k = 1 / math.cos(math.radians(lat))         # scale factor sec(phi): lengths grow by k, areas by k squared
    print(f"{name:<6} x = {x:8.1f} km   y = {y:7.1f} km   k = {k:.3f}   area x {k * k:.2f}")
Hanoi  x =  11770.0 km   y =  2392.8 km   k = 1.071   area x 1.15
